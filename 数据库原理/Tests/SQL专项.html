二、SQL编程题

编写SQL语句解决实际问题（20题）

📌 说明： 请根据题目要求编写SQL查询语句，使用上面提供的数据表结构

1. 使用集合查询，查询选修了C1或C2课程的学生学号

SELECT sno FROM SC WHERE cno = 'C1'
UNION
SELECT sno FROM SC WHERE cno = 'C2';

2. 使用集合查询，查询没有选修C1课程的学生学号

SELECT sno FROM Student
EXCEPT
SELECT sno FROM SC WHERE cno = 'C1';

3. 使用嵌套查询，查询比"李勇"年龄小的学生姓名

SELECT sname FROM Student
WHERE sage < (
    SELECT sage FROM Student WHERE sname = '李勇'
);

4. 使用嵌套查询，查询其他系中比计算机系(CS)所有学生年龄都大的学生姓名和年龄

SELECT sname, sage FROM Student
WHERE sdept <> 'CS' AND sage > ALL (
    SELECT sage FROM Student
    WHERE sdept = 'CS'
);

5. 创建新表 student_avg (sno char(5) primary key, avg_grade int)，汇总每位学生的平均成绩插入到新表中

INSERT INTO student_avg (sno, avg_grade)
SELECT sno, AVG(grade) FROM SC GROUP BY sno;

6. 创建新表 course_count (cno char(4) primary key, stu_count int)，汇总每门课程的选课人数插入到新表中

INSERT INTO course_count (cno, stu_count)
SELECT cno, COUNT(*) FROM SC GROUP BY cno;

7. 删除"李勇"的选课记录

DELETE FROM SC WHERE sno = (
    SELECT sno FROM Student WHERE sname = '李勇'
);

8. 删除"数据库"课程的选课记录

DELETE FROM SC WHERE cno = (
    SELECT cno FROM Course WHERE cname = '数据库'
);

9. 使用连接查询，将同时选修C1和C2两门课程的学生的成绩提高10%

UPDATE SC
SET grade = grade * 1.1
WHERE sno IN (
    SELECT s1.sno
    FROM SC s1, SC s2
    WHERE s1.sno = s2.sno AND s1.cno = 'C1' AND s2.cno = 'C2'
);

10. 使用连接子查询，将选修了"数据库"课程的学生的成绩改为0

UPDATE SC
SET grade = 0
WHERE cno = (
    SELECT cno FROM Course WHERE cname = '数据库'
);

11. 查询选修了3门以上课程的学生学号

SELECT sno FROM SC 
GROUP BY sno 
HAVING COUNT(*) > 3;

12. 查询平均成绩最高的课程号

SELECT cno FROM SC 
GROUP BY cno 
HAVING AVG(grade) >= ALL (
    SELECT AVG(grade) 
    FROM SC 
    GROUP BY cno
);

13. 查询选修了全部课程的学生姓名

SELECT sname FROM Student 
WHERE NOT EXISTS (
    SELECT * FROM Course 
    WHERE NOT EXISTS (
        SELECT * FROM SC 
        WHERE SC.sno = Student.sno AND SC.cno = Course.cno
    )
);

14. 查询每门课程的最高分、最低分和平均分

SELECT cno, MAX(grade), MIN(grade), AVG(grade) 
FROM SC 
GROUP BY cno;

15. 查询至少选修了学号为S1的学生所选修全部课程的学生学号

SELECT DISTINCT sno 
FROM SC SCX 
WHERE NOT EXISTS (
    SELECT * FROM SC SCY 
    WHERE SCY.sno = 'S1' AND NOT EXISTS (
        SELECT * FROM SC SCZ 
        WHERE SCZ.sno = SCX.sno AND SCZ.cno = SCY.cno
    )
);

16. 查询所有课程成绩都在85分以上的学生姓名

SELECT sname FROM Student 
WHERE NOT EXISTS (
    SELECT * FROM SC 
    WHERE SC.sno = Student.sno AND grade < 85
);

17. 查询每个学生选修课程的门数和平均成绩

SELECT sno, COUNT(*), AVG(grade) 
FROM SC 
GROUP BY sno;

18. 查询平均成绩大于80分的课程号

SELECT cno FROM SC 
GROUP BY cno 
HAVING AVG(grade) > 80;

19. 查询选修了"数据库"课程的学生的学号和成绩，按成绩降序排列

SELECT sno, grade 
FROM SC 
WHERE cno = (
    SELECT cno FROM Course WHERE cname = '数据库'
) 
ORDER BY grade DESC;

20. 查询每门课程成绩最高的学生学号和成绩

SELECT cno, sno, grade 
FROM SC SC1 
WHERE grade = (
    SELECT MAX(grade) 
    FROM SC SC2 
    WHERE SC1.cno = SC2.cno
);
